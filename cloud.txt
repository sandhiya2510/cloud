const API_KEY = "e18f6f8809c828c7d769e6ad18161521";
async function getWeather() {
    const city = document.getElementById("city").value;
    const API_URL =
        `https://api.openweathermap.org/data/2.5/weather?q=${city}&appid=${API_KEY}&units=metric`;

    try {
        const response = await fetch(API_URL);
        if (!response.ok) {
            throw new Error(`HTTP error! Status: ${response.status}`);
        }
        const data = await response.json();
        document.getElementById("result").innerHTML = `
            <h2>${data.name}</h2>
            <p>Temperature: ${data.main.temp} °C</p>
            <p>Feels Like: ${data.main.feels_like} °C</p>
            <p>Humidity: ${data.main.humidity}%</p>
            <p>Weather: ${data.weather[0].description}</p>
        `;
    } catch (error) {
        document.getElementById("result").innerHTML =
            "Unable to get weather data.";
        console.error(error);
    }
}

app.get("/api/products/:id", (req, res) => {
    const id = parseInt(req.params.id);
    const product = products.find(p => p.id === id);
    if (!product) {
        return res.json({ message: "Product not found" });
    }
    res.json(product);
});
app.post("/api/products", (req, res) => {
    const newProduct = {
        id: products.length + 1,
        name: req.body.name,
        category: req.body.category,
        price: req.body.price
    };
    products.push(newProduct);
    res.json(newProduct);
});
app.put("/api/products/:id", (req, res) => {
    const id = parseInt(req.params.id);
    const product = products.find(p => p.id === id);
    if (!product) {
        return res.json({ message: "Product not found" });
    }
    product.name = req.body.name;
    product.category = req.body.category;
    product.price = req.body.price;
    res.json(product);
});
app.delete("/api/products/:id", (req, res) => {
    const id = parseInt(req.params.id);
    const index = products.findIndex(p => p.id === id);
    if (index === -1) {
        return res.json({ message: "Product not found" });
    }
    const deletedProduct = products.splice(index, 1);
    res.json(deletedProduct);
});

    const name = document.getElementById("name").value.trim();

        await addDoc(collection(db, "students"), {
            name, regno, department, uid: user.uid

async function readStudents() {
    const user = auth.currentUser;
    if (!user) return;

    const q = query(
        collection(db, "students"),
        where("uid", "==", user.uid)
    );

    const result = await getDocs(q);
    const records = document.getElementById("records");
    records.innerHTML = "";

    result.forEach(item => {
        const s = item.data();
        const div = document.createElement("div");

        const details = document.createElement("p");
        details.textContent =
            "Name: " + s.name +
            " | Reg No: " + s.regno +
            " | Department: " + s.department;

        const edit = document.createElement("button");
        edit.textContent = "Edit";
        edit.onclick = () => editStudent(item.id, s);

        const del = document.createElement("button");
        del.textContent = "Delete";
        del.onclick = () => deleteStudent(item.id);

        div.append(details, edit, del);
        records.appendChild(div);
    });
}

from flask import Flask, render_template, request
import dropbox
import os

app = Flask(__name__)

ACCESS_TOKEN = ""

@app.route("/")
def home():
    return render_template("index.html")

@app.route("/upload", methods=["POST"])
def upload():
    file = request.files.get("file")

    if not file or file.filename == "":
        return "Please select a file!"

    try:
        dbx = dropbox.Dropbox(ACCESS_TOKEN)
        path = "/" + os.path.basename(file.filename)

        dbx.files_upload(
            file.read(),
            path,
            mode=dropbox.files.WriteMode.overwrite
        )

        return "File uploaded successfully to Dropbox: " + file.filename

    except Exception as e:
        return "Upload failed: " + str(e)

if __name__ == "__main__":
    app.run(debug=True)

#include <omp.h>
#include <iostream>
using namespace std;

#define THREAD_COUNT 5

int serial_sum(int n)
{
    int sum = 0;
    for (int i = 1; i <= n; i++)
        sum += i;
    return sum;
}

int parallel_sum(int n)
{
    int sum = 0;

    #pragma omp parallel for reduction(+:sum)
    for (int i = 1; i <= n; i++)
        sum += i;

    return sum;
}

int main()
{
    int n = 100000;

    cout << "Serial Sum: " << serial_sum(n) << endl;
    cout << "Parallel Sum: " << parallel_sum(n) << endl;

    return 0;
}
